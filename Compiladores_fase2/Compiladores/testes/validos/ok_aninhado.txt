algoritmo "Aninhado"
funcao dobro(n: inteiro): inteiro
inicio
   retorne n * 2
fimfuncao
funcao soma_dobros(a: inteiro, b: inteiro): inteiro
inicio
   retorne dobro(a) + dobro(b)
fimfuncao
procedimento mostrar(x: inteiro, y: inteiro)
inicio
   escreval(soma_dobros(x, y))
   escreval(x - y)
fimprocedimento
var
   media: real
   v: vetor[0..2] de inteiro
   i: inteiro
inicio
   mostrar(3, 10)
   escreval(dobro(dobro(5)) + dobro(1))
   para i de 0 ate 2 faca
      v[i] <- i * i
   fimpara
   v[v[1]] <- v[2] + v[v[1] + 1]
   escreval(v[1])
   media <- (v[0] + v[1] + v[2]) / 3
   escreval(media)
   escreval(-v[2] MOD 3)
