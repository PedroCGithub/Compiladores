algoritmo "Aninhado"
funcao dobro(n: inteiro): inteiro
inicio
   retorne n * 2
fimfuncao
funcao soma_dobros(a: inteiro, b: inteiro): inteiro
inicio
   retorne dobro(a) + dobro(b)
fimfuncao
procedimento mostrar(x: inteiro, y: inteiro)
inicio
   escreval(soma_dobros(x, y))
   escreval(x - y)
fimprocedimento
funcao fib(n: inteiro): inteiro
inicio
   se (n < 2) entao
      retorne n
   fimse
   retorne fib(n - 1) + fib(n - 2)
fimfuncao
var
   media: real
   v: vetor[0..2] de inteiro
   i: inteiro
inicio
   mostrar(3, 10)
   escreval(fib(10))
   para i de 0 ate 2 faca
      v[i] <- i * i
   fimpara
   media <- (v[0] + v[1] + v[2]) / 3
   escreval(media)
   escreval(-v[2] MOD 3)
fimalgoritmo
