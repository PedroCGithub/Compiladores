algoritmo "Recursao"
funcao fat(n: inteiro): inteiro
inicio
   se (n <= 1) entao
      retorne 1
   fimse
   retorne n * fat(n - 1)
fimfuncao
funcao fib(n: inteiro): inteiro
inicio
   se (n < 2) entao
      retorne n
   fimse
   retorne fib(n - 1) + fib(n - 2)
fimfuncao
funcao somar(a: inteiro, b: inteiro): inteiro
inicio
   retorne a + b
fimfuncao
procedimento mostrar(x: inteiro, y: inteiro)
inicio
   escreval(somar(fat(x), y))
fimprocedimento
var
   v: vetor[1..3] de inteiro
   i: inteiro
inicio
   escreval(fat(5))
   escreval(fib(10))
   mostrar(3, 100)
   para i de 1 ate 3 faca
      v[i] <- fat(i + 1)
   fimpara
   v[v[1]] <- v[3] MOD 5
   escreval(v[1], v[2], v[3])
fimalgoritmo
